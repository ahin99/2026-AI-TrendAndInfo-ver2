import ast<br>import time<br>import tracemalloc
# =========================
# Complexity Analyzer
# =========================
def analyze_complexity(code):
```plain text
tree = ast.parse(code)

loop_count = sum(
    isinstance(node, ast.For)
    for node in ast.walk(tree)
)

while_count = sum(
    isinstance(node, ast.While)
    for node in ast.walk(tree)
)

total_loops = loop_count + while_count

if total_loops == 0:
    return "O(1)"

elif total_loops == 1:
    return "O(n)"

elif total_loops == 2:
    return "O(n²)"

elif total_loops == 3:
    return "O(n³)"

else:
    return "복잡도 추정 불가"
```
# =========================
# Runtime Benchmark
# =========================
def benchmark(code):
```plain text
start = time.perf_counter()

exec(code, {})

end = time.perf_counter()

return end - start
```
# =========================
# Memory Profiler
# =========================
def measure_memory(code):
```plain text
tracemalloc.start()

exec(code, {})

current, peak = tracemalloc.get_traced_memory()

tracemalloc.stop()

return peak / 1024 / 1024
```
# =========================
# Accuracy Validator
# =========================
def validate(result, expected):
```plain text
if result == expected:
    return 100

return 0
```
# =========================
# Optimization Advisor
# =========================
def suggest(complexity):
```plain text
suggestions = {
    "O(n²)": "중첩 반복문 제거 또는 Hash Set 활용 고려",
    "O(n³)": "알고리즘 재설계 필요",
    "O(n)": "이진 탐색 적용 가능성 검토",
    "O(1)": "매우 효율적",
}

return suggestions.get(
    complexity,
    "분석 불가"
)
```
